# Optional portable native desktop build. The original Makefile/CLI is unchanged.
cmake_minimum_required(VERSION 3.21)
project(NAVALMDesktop VERSION 5.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FLTK_MSVC_RUNTIME_DLL OFF CACHE BOOL "" FORCE)
set(FLTK_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(opt TEST EXAMPLES FLUID FLTK_OPTIONS GL FORMS HTML_DOCS PDF_DOCS)
    set(FLTK_BUILD_${opt} OFF CACHE BOOL "" FORCE)
endforeach()
set(FLTK_USE_SYSTEM_LIBJPEG OFF CACHE BOOL "" FORCE)
set(FLTK_USE_SYSTEM_LIBPNG OFF CACHE BOOL "" FORCE)
set(FLTK_USE_SYSTEM_ZLIB OFF CACHE BOOL "" FORCE)
set(NAVALM_FLTK_SOURCE "" CACHE PATH "Existing FLTK 1.4.5 source directory")
if(NAVALM_FLTK_SOURCE)
    add_subdirectory("${NAVALM_FLTK_SOURCE}" fltk EXCLUDE_FROM_ALL)
else()
    include(FetchContent)
    FetchContent_Declare(fltk
        URL https://github.com/fltk/fltk/releases/download/release-1.4.5/fltk-1.4.5-source.tar.gz
        URL_HASH SHA256=eede1fb2b8e9c2e581e77082e15252145855c79aad30070ee3b24aabe2f926f1)
    FetchContent_MakeAvailable(fltk)
endif()
file(GLOB ASTRO CONFIGURE_DEPENDS libastro/*.c)
list(FILTER ASTRO EXCLUDE REGEX "/deltat.c$")
add_library(navastro STATIC ${ASTRO})
add_library(navnovas STATIC novas/novas.c novas/novascon.c novas/nutation.c novas/readeph0.c)
target_include_directories(navnovas PRIVATE novas)
target_compile_definitions(navnovas PRIVATE nutation=novas_nutation refract=novas_refract)
add_library(navcore STATIC navsight.c navrc.c nav_engine_sel.c nav_engine_x.c nav_engine_n.c
    nav_solsys440.c nav_eph440.c nav_stars_x.c)
target_include_directories(navcore PUBLIC . PRIVATE novas)
target_link_libraries(navcore PUBLIC navastro navnovas)
add_executable(navalm-desktop WIN32 navfltk.cpp navdownload.cpp)
target_link_libraries(navalm-desktop PRIVATE navcore fltk::fltk)
if(WIN32)
    enable_language(RC)
    target_sources(navalm-desktop PRIVATE tools/windows.rc)
    target_link_libraries(navalm-desktop PRIVATE winhttp)
    target_compile_definitions(navalm-desktop PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(navalm-desktop PROPERTIES OUTPUT_NAME NAVALM)
    if(MINGW)
        target_link_options(navalm-desktop PRIVATE -static)
    endif()
endif()
add_executable(desktop-core-check tests/desktop_core.cpp)
target_link_libraries(desktop-core-check PRIVATE navcore)
if(MINGW)
    target_link_options(desktop-core-check PRIVATE -static)
endif()
enable_testing()
add_test(NAME desktop-core COMMAND desktop-core-check)
set_tests_properties(desktop-core PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
option(NAVALM_EMBED_DATA "Embed the map and JPL ephemeris in the Windows exe" OFF)
set(NAVALM_EPHEMERIS_FILE "${CMAKE_SOURCE_DIR}/ephem/de440s.nav" CACHE FILEPATH "Repacked JPL file")
set(NAVALM_MAP_FILE "${CMAKE_SOURCE_DIR}/.local/maps/ne_10m_land.shp" CACHE FILEPATH "Natural Earth land shapefile")
if(WIN32 AND NAVALM_EMBED_DATA)
    if(NOT EXISTS "${NAVALM_EPHEMERIS_FILE}" OR NOT EXISTS "${NAVALM_MAP_FILE}")
        message(FATAL_ERROR "Embedded build needs ephem/de440s.nav and Natural Earth map. See WINDOWS.md.")
    endif()
    configure_file(cmake/windows-assets.rc.in windows-assets.rc @ONLY)
    target_sources(navalm-desktop PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows-assets.rc")
    target_compile_definitions(navalm-desktop PRIVATE NAVALM_EMBEDDED_DATA)
endif()
add_executable(mkde440 tools/mkde440.c nav_eph440.c)
target_include_directories(mkde440 PRIVATE .)

if(UNIX)
    target_link_libraries(mkde440 PRIVATE m)
endif()
